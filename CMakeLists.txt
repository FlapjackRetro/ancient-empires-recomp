cmake_minimum_required(VERSION 3.16)
project(empires C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(SDL2 REQUIRED)

file(GLOB GEN_SRC CONFIGURE_DEPENDS src/recomp/gen/*.c)
add_executable(empires
  src/main.c src/platform_sdl.c src/opl.cpp
  third_party/ymfm/ymfm_opl.cpp third_party/ymfm/ymfm_adpcm.cpp third_party/ymfm/ymfm_pcm.cpp
  src/recomp/runtime.c src/recomp/vga.c src/recomp/cpu.c
  ${GEN_SRC})
option(EMPIRES_TRACE "Record a ring of entered functions and dump it on a crash" OFF)
target_compile_definitions(empires PRIVATE RECOMP_IRQ RECOMP_MEM_HOOK)
if(EMPIRES_TRACE)
  target_compile_definitions(empires PRIVATE RECOMP_TRACE)
endif()
target_include_directories(empires PRIVATE src src/recomp)
if(MSVC)
  target_compile_options(empires PRIVATE /w /O2)
  target_compile_definitions(empires PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(empires PRIVATE -O2 -w)
endif()
if(TARGET SDL2::SDL2)
  target_link_libraries(empires PRIVATE SDL2::SDL2)
  if(TARGET SDL2::SDL2main AND WIN32)
    target_link_libraries(empires PRIVATE SDL2::SDL2main)
  endif()
else()
  target_include_directories(empires PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(empires PRIVATE ${SDL2_LIBRARIES})
endif()
